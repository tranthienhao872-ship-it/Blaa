	local function TweenModel(model, targetCF, duration)
		if not model or not model.Parent then return nil end
		local cfVal = Instance.new("CFrameValue")
		cfVal.Value = model:GetPivot()

		local conn
		conn = cfVal.Changed:Connect(function(val)
			if model and model.Parent then
				model:PivotTo(val)
			end
		end)

		local tween = TweenService:Create(cfVal, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut), {Value = targetCF})
		tween:Play()

		tween.Completed:Connect(function()
			if conn then conn:Disconnect() end
			cfVal:Destroy()
		end)
		return tween
	end

	local function GetGitSound(GithubSnd, SoundName)

		return sound

	local function GetGithubModel(url)
		local fileName = string.match(url, "([^/]+)$") or "temp_model.rbxm"

		if not isfile(fileName) then
			local response = http_request({
				Url = url,
				Method = "GET"
			})

			if not response or response.StatusCode ~= 200 then
				return nil
			end

			writefile(fileName, response.Body)
		end

		local assetId = getasset(fileName)
		local success, result = pcall(function()
			return game:GetObjects(assetId)[1]
		end)

		if success then
			return result
		end

		return nil
	end

	local function DrainCold(amount, normal)

		if character then
			local lighter = character:FindFirstChild("Lighter")

					return 
				end
			end
		end

		if activeLightFixture and character and character:FindFirstChild("HumanoidRootPart") then
			local hrp = character.HumanoidRootPart
			local fixturePos = activeLightFixture.LightFixture.Neon.Position
			if (hrp.Position - fixturePos).Magnitude <= 20 then				
				normal = math.abs(tonumber(normal) or 0) 
				SetCold(currentCold + normal)
			end
		end

		if coldDepleted then return end
		amount = math.max(0, tonumber(amount) or 0)
		SetCold(currentCold - amount)
	end
