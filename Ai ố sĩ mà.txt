	local function GetGitSound(GithubSnd, SoundName)

	local function GetGithubModel(url)
		local fileName = string.match(url, "([^/]+)$") or "temp_model.rbxm"

		if not isfile(fileName) then
			local response = http_request({
				Url = url,
				Method = "GET"
			})

			if not response or response.StatusCode ~= 200 then
				return nil
			end

			writefile(fileName, response.Body)
		end

		local assetId = getasset(fileName)
		local success, result = pcall(function()
			return game:GetObjects(assetId)[1]
		end)

		if success then
			return result
		end

		return nil
	end

		if newValue > lowestCold then newValue = lowestCold end
		if newValue < lowestCold then lowestCold = newValue end

							AngleY = angleY

				highlight.Adornee = spikeGroup
				highlight.FillColor = Color3.fromRGB(173, 221, 255)

				local lockTime = 0

				local initialPos = entityPart.Position
				local initialTarget = hrp.Position

				connection = RunService.RenderStepped:Connect(function(dt)

					local travelTime = spikeTimer - lockTime

							item.Part.CFrame = currentBaseCF * CFrame.new(item.OffsetX, 0, 0) * CFrame.Angles(-90, math.rad(item.AngleY), 0)

		local RENDER_NAME = "DOORS_FOV_Override"

		local function FovKit2(targetFov, normalFov, speedOut)
			targetFov = targetFov or 100
			normalFov = normalFov or workspace.CurrentCamera.FieldOfView
			speedOut = speedOut or 0.5

			pcall(function()
				RunService:UnbindFromRenderStep(RENDER_NAME)
			end)

			local camera = workspace.CurrentCamera
			if not camera then return end

			camera.FieldOfView = targetFov

			local startTime = os.clock()

			RunService:BindToRenderStep(RENDER_NAME, Enum.RenderPriority.Camera.Value + 100, function()
				local elapsed = os.clock() - startTime
				local alpha = math.clamp(elapsed / speedOut, 0, 1)
				local easedAlpha = TweenService:GetValue(alpha, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

				camera.FieldOfView = targetFov + (normalFov - targetFov) * easedAlpha

				if alpha >= 1 then
					camera.FieldOfView = normalFov
					RunService:UnbindFromRenderStep(RENDER_NAME)
				end
			end)
		end
